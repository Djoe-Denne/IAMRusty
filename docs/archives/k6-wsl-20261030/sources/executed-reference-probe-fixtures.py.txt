def elf_fixture():
    raw = bytearray(400)
    raw[:64] = struct.pack("<16sHHIQQQIHHHHHH", b"\x7fELF\x02\x01\x01" + bytes(9), 3, 62, 1, 0, 64, 0, 0, 64, 56, 3, 0, 0, 0)
    interp = LOADER.encode() + b"\0"
    for index, values in enumerate(((1, 5, 0, 0, 0, 400, 400, 4096),
                                    (3, 4, 232, 232, 0, len(interp), len(interp), 1),
                                    (2, 4, 280, 280, 0, 64, 64, 8))):
        raw[64 + index * 56:120 + index * 56] = struct.pack("<IIQQQQQQ", *values)
    raw[232:232 + len(interp)] = interp
    for index, values in enumerate(((5, 360), (10, 11), (1, 1), (0, 0))):
        raw[280 + index * 16:296 + index * 16] = struct.pack("<QQ", *values)
    raw[360:371] = b"\0liba.so.1\0"
    return bytes(raw)

@contextlib.contextmanager
def posix_unlink_fixture(root):
    """Windows readonly-file unlink differs from POSIX. Only disposable files
    beneath this known test directory may have that OS boundary simulated."""
    real_unlink = Path.unlink
    def unlink(path, *args, **kwargs):
        if path.is_relative_to(root) and path.is_file():
            path.chmod(0o600)
        return real_unlink(path, *args, **kwargs)
    with mock.patch.object(Path, "unlink", unlink):
        yield


class ProbeTests(unittest.TestCase):
    def setUp(self):

    def test_strict_ldd_missing_with_exit_zero_is_not_zero(self):
        self.assertEqual(probe.ldd_count(GOOD, b"", 0, {"liba.so.1"}, {LOADER}), 0)
        missing = GOOD.replace(b"/lib/x86_64-linux-gnu/liba.so.1 (0x5678)", b"not found")
        self.assertEqual(probe.ldd_count(missing, b"", 0, {"liba.so.1"}, {LOADER}), 1)

    def test_ldd_empty_partial_unknown_or_nonzero_never_means_zero(self):
        for raw, err, code in ((b"", b"", 0), (b"\n", b"", 0), (GOOD[:-1], b"", 0), (GOOD, b"", 1),
                               (GOOD + b"unknown\n", b"", 0), (GOOD, b"warning", 0), (b"\xff\n", b"", 0),
                               (GOOD.replace(b"\t/lib64/ld-linux-x86-64.so.2 (0xabcd)\n", b""), b"", 0),
                               (GOOD + GOOD, b"", 0)):
            self.assertIsNone(probe.ldd_count(raw, err, code, {"liba.so.1"}, {LOADER}))
        self.assertIsNone(probe.ldd_count(GOOD, b"", 0, {"liba.so.1", "libb.so.2"}, {LOADER}))

    def test_privacy_overflow_and_identity_missing_withhold_count_and_payload(self):
        for raw in (GOOD + b"CONFIDENTIAL_SENTINEL", GOOD + b"https://example.invalid", GOOD + b"\xff", GOOD + b"x" * safety.LIMIT):
            record, _ = self.launch(raw)
            projected = probe.project_probe(record, {"liba.so.1"}, {LOADER}, "ldd")
            self.assertIsNone(projected["missing"])
            self.assertNotEqual(record["privacy"], "PASS")
            self.assertNotIn("CONFIDENTIAL_SENTINEL", json.dumps(projected))
        record, _ = self.launch(cleanup=False)
        self.assertEqual(record["cleanup"], "FAIL")
        self.assertIsNone(probe.project_probe(record, {"liba.so.1"}, {LOADER}, "ldd")["missing"])
        record, _ = self.launch()
        record["cleanup"] = "INCONCLUSIVE"
        self.assertIsNone(probe.project_probe(record, {"liba.so.1"}, {LOADER}, "ldd")["missing"])
        child = safety.Child()
        child.stdout, child.code, child.cleanup = GOOD, 0, True
        with mock.patch.object(runner, "Child", return_value=child), mock.patch.object(child, "execute", return_value=0):
            record = probe.run_one(SimpleNamespace(runner=runner, safety=safety), "ldd", Path("/private/run"),
                                   PurePosixPath("/verified/chrome"), PurePosixPath("/private/probe.py"))

        self.assertIsNone(probe.project_probe(record, {"liba.so.1"}, {LOADER}, "ldd")["missing"])

    def test_retirement_tracks_both_real_child_artifact_shapes_and_replacement(self):

    def test_launch_fixed_order_current_projection_and_fail_closed_negatives(self):
        for fault in ("none", "missing", "privacy", "identity", "failed_zero_exit"):
            with self.subTest(fault=fault), scratch() as root, posix_unlink_fixture(root), \
                    mock.patch.dict(os.environ, {}, clear=True), contextlib.ExitStack() as stack:
                source, tools, output = root / "source", root / "tools", root / "out"
                source.mkdir(); tools.mkdir()
                chrome = tools / "chrome"; chrome.write_bytes(elf_fixture())
                api = SimpleNamespace(runner=SimpleNamespace(pinned_tools=mock.Mock(return_value=({"k6": chrome, "chromium": chrome}, {})),
                                      verify_tools=mock.Mock()), recheck=mock.Mock(),
                                      safety=SimpleNamespace(private_path=lambda p, **k: p, name_chain=lambda p: (str(p),),
                                                            strict_json=safety.strict_json, confidential=safety.confidential))
                buffer, calls = io.BytesIO(), []
                def run(_api, kind, directory, _chrome, _script):
                    calls.append(kind)
                    self.assertEqual(directory, output)
                    (directory / "private" / ("children-" + str(len(calls)) * 32 + ".json")).write_bytes(b"[]")
                    (directory / "private/process-identities.jsonl").write_bytes(b"known private fixture\n")
                    raw = json.dumps(context_fixture(True)).encode() if kind == "context" else GOOD if kind == "ldd" else b""
                    record = {"stdout": raw, "stderr": b"", "privacy": "PASS", "cleanup": "PASS", "exit": 0,
                              "termination": "EXITED", "complete": True, "error": "NONE"}
                    if kind == "ldd" and fault == "missing":
                        record["stdout"] = GOOD.replace(b"/lib/x86_64-linux-gnu/liba.so.1 (0x5678)", b"not found")
                    if kind == "ldd" and fault == "privacy":
                        record.update(stdout=b"CONFIDENTIAL_SENTINEL", privacy="FAIL")
                    if kind == "unshare" and fault == "identity":
                        record["cleanup"] = "INCONCLUSIVE"
                    if kind == "unshare" and fault == "failed_zero_exit":
                        record.update(complete=False, error="CHILD_FAILED")
                    return record
                patches = (mock.patch.object(probe, "native_home", return_value=root.stat().st_dev),
                           mock.patch.object(probe, "private_chain", side_effect=lambda p, **k: Path(p)),
                           mock.patch.object(probe, "load_api", return_value=api),
                           mock.patch.object(probe, "read_small", return_value=(CANON / "config/tools.lock.json").read_bytes()),
                           mock.patch.object(probe, "trusted_system", return_value=()),
                           mock.patch.object(Path, "resolve", autospec=True, side_effect=lambda p, **k: p),
                           mock.patch.object(probe.os, "getuid", return_value=0, create=True),
                           mock.patch.object(probe, "collect_context", return_value=context_fixture()),
                           mock.patch.object(probe, "run_one", side_effect=run),
                           mock.patch.object(sys, "stdout", SimpleNamespace(buffer=buffer)))
                for patch in patches:
                    stack.enter_context(patch)
                code = probe.main(["--source-dir", str(source), "--tools-dir", str(tools), "--output", str(output)])
                value = json.loads(buffer.getvalue())
                self.assertEqual(calls, ["context", "unshare", "ldd"])
                self.assertEqual(code, 0 if fault == "none" else 2)
                self.assertEqual(value["retirement"], "PASS")
                self.assertFalse(output.exists())
                self.assertTrue(chrome.exists())
                self.assertNotIn(b"CONFIDENTIAL_SENTINEL", buffer.getvalue())
                self.assertEqual(value["HTTP"], "INCONCLUSIVE")
                self.assertEqual(value["Chromium"], "INCONCLUSIVE")
                if fault in {"privacy", "identity"}:
                    self.assertEqual(value["publication"], "WITHHELD")
                elif fault == "missing":
                    self.assertEqual(value["ldd"]["missing"], 1)
                elif fault == "failed_zero_exit":
                    self.assertEqual(value["unshare"]["outcome"], "INCONCLUSIVE")
                else:
                    self.assertEqual(value["a_commands"], "NOT_EXECUTED")
                    self.assertEqual(value["comparison"]["kind"], "CONTEXTS_ONLY")
