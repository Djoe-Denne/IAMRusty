"""One-off native administration of the explicitly owned AIForAll WSL host; no browser/test execution."""
import json
import os
from pathlib import Path
import re
import subprocess
from urllib.parse import urlsplit

if os.geteuid() != 0:
    raise SystemExit("OWNED_HOST_ADMIN_REQUIRES_ROOT")
release = dict(line.split("=", 1) for line in Path("/etc/os-release").read_text().splitlines() if "=" in line)
if release.get("ID", "").strip('"') != "ubuntu" or release.get("VERSION_ID", "").strip('"') != "24.04":
    raise SystemExit("OWNED_HOST_PLATFORM_REFUSED")

sources = sorted(Path("/etc/apt/sources.list.d").glob("*.sources"))
if not sources:
    raise SystemExit("OFFICIAL_APT_SOURCES_MISSING")
classic = Path("/etc/apt/sources.list")
if classic.exists() and any(line.strip() and not line.lstrip().startswith("#") for line in classic.read_text().splitlines()):
    raise SystemExit("CLASSIC_APT_SOURCES_REQUIRE_REVIEW")
if any(Path("/etc/apt/sources.list.d").glob("*.list")):
    raise SystemExit("EXTRA_APT_SOURCES_REQUIRE_REVIEW")
hosts = set()
for source in sources:
    text = source.read_text()
    if re.search(r"(?im)^Trusted:\s*yes\s*$", text):
        raise SystemExit("UNAUTHENTICATED_SOURCE_REFUSED")
    for paragraph in re.split(r"\n\s*\n", text):
        if not paragraph.strip() or not any(line.startswith("URIs:") for line in paragraph.splitlines()):
            continue
        fields = dict(line.split(":", 1) for line in paragraph.splitlines() if ":" in line and not line.startswith("#"))
        if fields.get("Signed-By", "").strip() != "/usr/share/keyrings/ubuntu-archive-keyring.gpg":
            raise SystemExit("APT_SIGNING_KEY_REQUIRE_REVIEW")
        for value in fields.get("URIs", "").split():
            uri = urlsplit(value)
            if uri.scheme not in {"http", "https"} or uri.hostname not in {"archive.ubuntu.com", "security.ubuntu.com", "ports.ubuntu.com"} or uri.username or uri.password:
                raise SystemExit("NON_UBUNTU_SOURCE_REFUSED")
            hosts.add(uri.hostname)
        if any(suite not in {"noble", "noble-updates", "noble-backports", "noble-security"} for suite in fields.get("Suites", "").split()):
            raise SystemExit("OTHER_UBUNTU_RELEASE_REFUSED")

packages = [
    "ca-certificates", "curl", "git", "python3", "binutils", "file",
    "libnspr4", "libnss3", "libasound2t64", "libatk1.0-0t64",
    "libatk-bridge2.0-0t64", "libatspi2.0-0t64", "libcups2t64",
    "libdbus-1-3", "libdrm2", "libexpat1", "libgbm1", "libglib2.0-0t64",
    "libx11-6", "libxcomposite1", "libxdamage1", "libxext6", "libxfixes3",
    "libxrandr2", "libxkbcommon0", "libcairo2", "libpango-1.0-0", "fonts-liberation",
]
root = Path("/var/log/aiforall-provision-20261030-mvuqe1yi")
root.mkdir(mode=0o700)
os.umask(0o077)
results = []
for phase, args in [
    ("refresh", ["/usr/bin/apt-get", "update"]),
    ("simulate", ["/usr/bin/apt-get", "--simulate", "--no-install-recommends", "install", *packages]),
    ("install", ["/usr/bin/apt-get", "--yes", "--no-install-recommends", "install", *packages]),
]:
    with (root / (phase + ".log")).open("xb") as log:
        completed = subprocess.run(args, stdout=log, stderr=subprocess.STDOUT, timeout=1200)
    results.append({"phase": phase, "exit": completed.returncode})
    if completed.returncode:
        print(json.dumps({"bootstrap": "FAILED", "results": results, "raw_logs_published": False, "browser_executed": False}))
        raise SystemExit(completed.returncode)

query = subprocess.run(["/usr/bin/dpkg-query", "-W", "--showformat=${binary:Package}\t${Version}\n", *packages], capture_output=True, text=True, check=True, timeout=20)
manifest = {line.split("\t", 1)[0]: line.split("\t", 1)[1] for line in query.stdout.splitlines()}
(root / "packages.json").write_text(json.dumps(manifest, sort_keys=True) + "\n")
print(json.dumps({"bootstrap": "PACKAGES_INSTALLED", "results": results, "requested_package_count": len(packages), "requested_packages": manifest, "source_hosts": sorted(hosts), "raw_logs_published": False, "browser_executed": False}, sort_keys=True))
