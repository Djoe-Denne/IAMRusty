function Refuses([scriptblock]$Action){
    $bad=$false
    try{& $Action|Out-Null}catch{
        $bad=$true
        if($_.Exception.Message -cne 'PROBE_PARENT_REFUSED'){[Console]::WriteLine(('unexpected_refusal_id='+$_.FullyQualifiedErrorId))}
        Check ($_.Exception.Message -ceq 'PROBE_PARENT_REFUSED')
    }
    Check $bad
}

}
function Lease {

function Open-LddQuota {
    # Only the OS file-opening boundary is redirected to disposable test files.
    # Get-LddQuotaPath itself is unchanged and must remain fixed/canonical.
    $script:OpenCalls++
    $s=[IO.File]::Open($script:FixtureQuota,[IO.FileMode]::CreateNew,[IO.FileAccess]::ReadWrite,[IO.FileShare]::None)
    $opened.Add($s)
    if($script:Fault -ceq 'none'){return $s}
    $wrapper=[pscustomobject]@{Real=$s;Fault=$script:Fault}
    $wrapper|Add-Member ScriptMethod Write {param($raw,$offset,$length)
        if($this.Fault -ceq 'write'){throw 'KNOWN_FIXTURE_WRITE_FAILURE'}
        $this.Real.Write($raw,$offset,$length)
    }
    $wrapper|Add-Member ScriptMethod Flush {param($disk) throw 'KNOWN_FIXTURE_FLUSH_FAILURE'}
    $wrapper|Add-Member ScriptMethod Dispose {$this.Real.Dispose()}
    return $wrapper
}
