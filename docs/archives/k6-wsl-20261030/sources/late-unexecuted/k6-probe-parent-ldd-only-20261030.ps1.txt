function Open-LddQuota {
    # Fixed across modes, leases and hosts. Never remove/reset this file.
    return [IO.File]::Open((Get-LddQuotaPath),[IO.FileMode]::CreateNew,[IO.FileAccess]::ReadWrite,[IO.FileShare]::None)
}
function Assert-LddLease($Lease) {
    Need ($Lease.mission -ceq $Mission -and $Lease.status -ceq 'active' -and $Lease.selection -ceq 'ldd-only' -and
        $Lease.probe_sha256 -ceq $ProbeSha -and $Lease.lease_id -cmatch '^[a-f0-9]{32}$')
    $started=[datetime]::ParseExact($Lease.started_utc,'o',[Globalization.CultureInfo]::InvariantCulture,[Globalization.DateTimeStyles]::RoundtripKind)
    $age=([datetime]::UtcNow-$started).TotalSeconds
    Need ($started.Kind -eq [DateTimeKind]::Utc -and $age -ge 0 -and $age -le 600)
}
function Reserve-LddException($Lease) {
    Assert-LddLease $Lease
    $entry=@{v=1;mission=$Mission;selection='ldd-only';lease_id=$Lease.lease_id;probe_sha256=$ProbeSha;
        previous_ldd_calls=1;additional_ldd_reserved=1;created_utc=[datetime]::UtcNow.ToString('o')}
    $raw=$Utf8.GetBytes(($entry|ConvertTo-Json -Compress))
    Need ($raw.Length -gt 0 -and $raw.Length -le 4096)
    $stream=$null
    try {
        $stream=Open-LddQuota
        $stream.Write($raw,0,$raw.Length)
        $stream.Flush($true)
        # Hold the same exclusive file handle through the attempted phase. On
        # Windows FileShare.None also prevents rename/delete substitution while
        # held. Closing on any failure never deletes or releases the reservation.
        return [pscustomobject]@{Path=(Get-LddQuotaPath);Stream=$stream;SafeHandle=$stream.SafeFileHandle;
            Handle=$stream.SafeFileHandle.DangerousGetHandle();Raw=$raw;LeaseId=$Lease.lease_id;Source=$ProbeSha;Attempted=$false}
    } catch {
        if($null -ne $stream){$stream.Dispose()}
        throw 'PROBE_PARENT_REFUSED'
    }
}
function Assert-LddReservation($Slot,$Lease,[string]$Selected) {
    Need ($Selected -ceq 'ldd-only' -and $null -ne $Slot)
    Assert-LddLease $Lease
    Need ($Slot.Path -ceq (Get-LddQuotaPath) -and $Slot.LeaseId -ceq $Lease.lease_id -and $Slot.Source -ceq $ProbeSha -and !$Slot.Attempted)
    $s=$Slot.Stream
    Need ($s.CanRead -and $s.CanWrite -and !$s.SafeFileHandle.IsClosed -and !$s.SafeFileHandle.IsInvalid -and
        [object]::ReferenceEquals($s.SafeFileHandle,$Slot.SafeHandle) -and $s.SafeFileHandle.DangerousGetHandle() -eq $Slot.Handle)
    Need ($s.Length -eq $Slot.Raw.Length -and $s.Length -le 4096)
    $s.Position=0;$copy=New-Object byte[] $Slot.Raw.Length;$offset=0
    while($offset -lt $copy.Length){$n=$s.Read($copy,$offset,$copy.Length-$offset);Need ($n -gt 0);$offset+=$n}
    Need ([Convert]::ToBase64String($copy) -ceq [Convert]::ToBase64String($Slot.Raw))
}
function Get-LddLinuxArguments([string]$Selected,[string]$Probe,[string]$Source,[string]$Tools,[string]$Output) {

}
function Assert-LddPhase($Slot,$Lease,[string]$Selected,[string[]]$Actual,[string[]]$Expected) {
    Assert-LddReservation $Slot $Lease $Selected
    Need ($Actual.Count -eq $Expected.Count)
    for($i=0;$i -lt $Actual.Count;$i++){Need ($Actual[$i] -ceq $Expected[$i])}
}
function Begin-LddPhase($Slot,$Lease,[string]$Selected,[string[]]$Actual,[string[]]$Expected) {

    $Lease.ldd_launch='ATTEMPTED_OR_UNCERTAIN'
}

function Invoke-OwnedWsl([string[]]$LinuxArgs,[string]$Phase) {

    $start=[Diagnostics.ProcessStartInfo]::new();$start.FileName=$Wsl;$start.Arguments=$argv -join ' '
    $start.UseShellExecute=$false;$start.CreateNoWindow=$true
    $start.RedirectStandardInput=$true;$start.RedirectStandardOutput=$true;$start.RedirectStandardError=$true
    $start.EnvironmentVariables.Clear()
    foreach($name in @('SystemRoot','WINDIR','TEMP','TMP')){
        $value=[Environment]::GetEnvironmentVariable($name);Need (![string]::IsNullOrEmpty($value));$start.EnvironmentVariables[$name]=$value
    }
    $start.EnvironmentVariables['PATH']=Join-Path $env:SystemRoot 'System32'
    if($Phase -ceq 'ldd'){
        Begin-LddPhase $slot $lease $Mode $LinuxArgs $expectedLdd
        Save
    }
    $p=[Diagnostics.Process]::new();$p.StartInfo=$start;Need ($p.Start())
    $entry=@{phase=$Phase;pid=$p.Id;joined=$false;identity_reverified=$false;identity_observed_before=$false}
    $lease.invocations+=,$entry
    try{
        $handle=$p.Handle;$first=[K6Parent.Native]::Observe($handle)
        Need ($first.Pid -eq $p.Id -and $first.Created -gt 0 -and $first.Sid -ceq $sid -and $first.Image -ieq $Wsl)
    }catch{Save;Event ('ldd_parser_'+$Phase+'_created_unobserved');throw}
    $entry.created=$first.Created;$entry.sid=$first.Sid;$entry.image=$first.Image;$entry.identity_observed_before=$true
    Save;Event ('ldd_parser_'+$Phase+'_observed')
    $p.StandardInput.Close()
    $out=[K6Parent.FrameReader]::new($p.StandardOutput.BaseStream)
    $err=[K6Parent.FrameReader]::new($p.StandardError.BaseStream)
    $clock=[Diagnostics.Stopwatch]::StartNew();$line=$null;$clean=$true;$joined=$false
    while($clock.Elapsed.TotalSeconds -lt 120) {
        try{
            $value=$out.Poll(10)
            if($null -ne $value){if($null -ne $line){$clean=$false}else{$line=$value}}
            [void]$err.Poll(2)
            if($err.Seen -gt 0 -or $out.Seen -gt 65536 -or $err.Seen -gt 65536){$clean=$false}
        }catch{$clean=$false;try{$p.StandardOutput.Close();$p.StandardError.Close()}catch{}}
        if($p.WaitForExit(0) -and (!$clean -or ($out.Ended -and $err.Ended))){$joined=$true;break}
        Start-Sleep -Milliseconds 5
    }
    $entry.joined=$joined;$entry.capture_clean=$clean;Save
    Need $joined
    # Same owned handle; SID/image were observed before exit, not freshly re-read.
    Need ($p.Handle -eq $handle -and $p.HasExited -and $p.Id -eq $first.Pid -and
        $p.StartTime.ToFileTimeUtc() -eq $first.Created -and $p.ExitTime.ToFileTimeUtc() -ge $first.Created)
    $entry.exit=$p.ExitCode;$entry.identity_reverified=$true
    $entry.identity_method='retained_handle_pid_creation_exit'

    $code=$p.ExitCode;$p.Dispose();Save;Event ('ldd_parser_'+$Phase+'_joined')
    Need ($clean -and $null -ne $reply)

}
