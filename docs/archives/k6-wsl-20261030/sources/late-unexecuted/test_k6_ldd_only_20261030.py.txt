"""Pure selection/Child fixtures. No Linux or application process is launched."""
import ast
import contextlib
import hashlib
import importlib.util
import io
import json
import os
from pathlib import Path
import sys
from types import SimpleNamespace
import unittest
from unittest import mock

HERE = Path(__file__).resolve().parent
REFERENCES = {"k6-linux-probes-20261030.py": "048d9c9608409610b2ff989f6052c7e272e09cdcf75c0ae047516ee658bc3083",
              "test_k6_linux_probes_20261030.py": "0d0f230aa7ed23a8d394072255b700b9dcdd4c292b1c3ed0cacb71fb3d7afb20"}
REFERENCE_DIR = next((directory for directory in (HERE, HERE / "k6-native-probes-candidate-048d9c96")
                      if all((directory / name).is_file() and hashlib.sha256((directory / name).read_bytes()).hexdigest() == digest
                             for name, digest in REFERENCES.items())), None)
if REFERENCE_DIR is None:
    raise RuntimeError("REFERENCE_UNAVAILABLE")
REFERENCE_SPEC = importlib.util.spec_from_file_location("ldd_selection_reference", REFERENCE_DIR / "test_k6_linux_probes_20261030.py")
reference = importlib.util.module_from_spec(REFERENCE_SPEC)
REFERENCE_SPEC.loader.exec_module(reference)
SPEC = importlib.util.spec_from_file_location("ldd_only_candidate", HERE / "k6-linux-probes-ldd-only-20261030.py")
probe = importlib.util.module_from_spec(SPEC)
SPEC.loader.exec_module(probe)
safety, runner = reference.safety, reference.runner


class SelectionTests(unittest.TestCase):
    def simulate(self, payload=reference.GOOD, code=0, cleanup=True, exception=None, no_identity=False, argv=None, early=None):
        with reference.scratch() as root, reference.posix_unlink_fixture(root), \
                mock.patch.dict(os.environ, {}, clear=True), contextlib.ExitStack() as stack:
            source, tools, output = root / "source", root / "tools", root / "out"
            source.mkdir(); tools.mkdir()
            chrome = tools / "chrome"; chrome.write_bytes(reference.elf_fixture())
            api = SimpleNamespace(runner=runner, recheck=mock.Mock(), safety=SimpleNamespace(
                private_path=lambda p, **k: p, name_chain=lambda p: (str(p),),
                strict_json=safety.strict_json, confidential=safety.confidential))
            console, calls, executions = io.BytesIO(), [], []
            original = probe.run_one
            def command(actual_api, kind, directory, binary, script):
                calls.append(kind)
                self.assertEqual(kind, "ldd")
                with reference.fake_child_reads(payload, 0, cleanup, native_exit=code) as child, \
                        mock.patch.object(runner, "Child", return_value=child):
                    execute = child.execute
                    def boundary(args, **kwargs):
                        executions.append((args, kwargs))
                        if exception is not None:
                            raise exception
                        result = execute(args, **kwargs)
                        if no_identity:
                            child.owned.clear(); child.recorded.clear()
                        return result
                    with mock.patch.object(child, "execute", side_effect=boundary):
                        return original(actual_api, kind, directory, binary, script)
            patches = (mock.patch.object(probe, "native_home", return_value=root.stat().st_dev),
                       mock.patch.object(probe, "private_chain", side_effect=lambda p, **k: Path(p)),
                       mock.patch.object(probe, "load_api", return_value=api, side_effect=early),
                       mock.patch.object(probe, "read_small", return_value=(reference.CANON / "config/tools.lock.json").read_bytes()),
                       mock.patch.object(probe, "trusted_system", return_value=()),
                       mock.patch.object(Path, "resolve", autospec=True, side_effect=lambda p, **k: p),
                       mock.patch.object(probe.os, "getuid", return_value=0, create=True),
                       mock.patch.object(runner, "pinned_tools", return_value=({"k6": chrome, "chromium": chrome}, {})),
                       mock.patch.object(runner, "verify_tools"), mock.patch.object(probe, "run_one", side_effect=command),
                       mock.patch.object(sys, "stdout", SimpleNamespace(buffer=console)))
            for patch in patches:
                stack.enter_context(patch)
            arguments = argv if argv is not None else ["--selection", "ldd-only", "--source-dir", str(source),
                                                       "--tools-dir", str(tools), "--output", str(output)]
            result = probe.main(arguments)
            raw = console.getvalue(); safety.confidential(raw)
            value = json.loads(raw)
            self.assertNotIn(b"CONFIDENTIAL_SENTINEL", raw)
            self.assertEqual(value.get("unshare"), probe.empty_result()["unshare"])
            self.assertEqual(value["a_commands"], "NOT_EXECUTED")
            for name in ("a_context", "b_context", "comparison"):
                self.assertIsNone(value[name])
            self.assertEqual(value["HTTP"], "INCONCLUSIVE")
            self.assertEqual(value["Chromium"], "INCONCLUSIVE")
            self.assertTrue(chrome.exists())
            if value["retirement"] == "PASS":
                self.assertFalse(output.exists())
            for args, kwargs in executions:
                self.assertEqual(args, ["/usr/bin/ldd", str(chrome)])
                self.assertEqual(kwargs["timeout"], 10)
                self.assertIs(kwargs["browser"], False)
                self.assertEqual(kwargs["env"], safety.child_environment(output))
            return result, value, calls, executions

    def test_zero_is_success_for_only_the_selected_command(self):
        code, value, calls, executions = self.simulate()
        self.assertEqual(code, 0)
        self.assertEqual(value["ldd"], {"outcome": "ZERO_EXIT", "exit": 0, "termination": "EXITED", "missing": 0})
        self.assertEqual(calls, ["ldd"])
        self.assertEqual(len(executions), 1)

    def test_positive_count_preserves_native_zero_but_script_is_nonzero(self):
        missing = reference.GOOD.replace(b"/lib/x86_64-linux-gnu/liba.so.1 (0x5678)", b"not found")
        code, value, calls, _ = self.simulate(missing)
        self.assertEqual(code, 2)
        self.assertEqual(value["ldd"], {"outcome": "ZERO_EXIT", "exit": 0, "termination": "EXITED", "missing": 1})
        self.assertEqual(value["reason"], "DIAGNOSTIC_INCOMPLETE")
        self.assertEqual(calls, ["ldd"])

    def test_unknown_native_nonzero_and_signal_do_not_succeed(self):
        for payload, native in ((b"", 0), (b"unknown\n", 0), (reference.GOOD, 1), (reference.GOOD, -6)):
            with self.subTest(native=native, empty=not payload):
                code, value, calls, _ = self.simulate(payload, native)
                self.assertEqual(code, 2)
                self.assertIsNone(value["ldd"]["missing"])
                self.assertEqual(value["ldd"]["exit"], native)
                self.assertEqual(value["ldd"]["termination"], "SIGNAL" if native < 0 else "EXITED")
                self.assertEqual(calls, ["ldd"])

    def test_privacy_overflow_and_unproven_cleanup_are_nonzero(self):
        for payload, cleanup, identity in ((b"CONFIDENTIAL_SENTINEL", True, False), (b"\xff", True, False),
                                           (b"x" * (safety.LIMIT + 1), True, False),
                                           (reference.GOOD, False, False), (reference.GOOD, True, True)):
            with self.subTest(cleanup=cleanup, no_identity=identity, size=len(payload)):
                code, value, calls, _ = self.simulate(payload, cleanup=cleanup, no_identity=identity)
                self.assertEqual(code, 2)
                self.assertEqual(value["publication"], "WITHHELD")
                self.assertEqual(calls, ["ldd"])

    def test_timeout_interruption_and_early_guard_do_not_add_commands(self):
        for error in (safety.Rejected("CHILD_TIMEOUT"), safety.Rejected("INTERRUPTED"), KeyboardInterrupt()):
            with self.subTest(error=type(error).__name__):
                code, value, calls, executions = self.simulate(exception=error)
                self.assertEqual(code, 2)
                self.assertEqual(calls, ["ldd"])
                self.assertEqual(len(executions), 1)
                self.assertNotEqual(value["cleanup"], "PASS")
        code, _, calls, executions = self.simulate(early=probe.Refused("SOURCE_UNTRUSTED"))
        self.assertEqual(code, 2); self.assertEqual(calls, []); self.assertEqual(executions, [])

    def test_selection_is_explicit_closed_and_no_full_or_context_entry_exists(self):
        for argv in ([], ["--context-child"], ["--selection", "run"], ["--selection", "ldd-only"],
                     ["--source-dir", "x", "--tools-dir", "x", "--output", "x"],
                     ["--selection", "ldd-only", "--source-dir", "x", "--tools-dir", "x", "--output", "x", "extra"]):
            with self.subTest(argv=argv):
                code, _, calls, executions = self.simulate(argv=argv)
                self.assertEqual(code, 2); self.assertEqual(calls, []); self.assertEqual(executions, [])
        for kind in ("context", "unshare"):
            with mock.patch.object(runner, "Child") as child, self.assertRaises(probe.Refused):
                probe.run_one(SimpleNamespace(runner=runner), kind, Path("/private"), Path("/chrome"), Path("/probe"))
            child.assert_not_called()

    def test_public_predicate_rejects_old_userns_proof_and_boolean_count(self):
        value = probe.empty_result()
        value.update(privacy="PASS", cleanup="PASS", retirement="PASS", ldd={"outcome": "ZERO_EXIT", "exit": 0,
                                                                          "termination": "EXITED", "missing": 0})
        self.assertTrue(probe.selection_complete(value))
        value["unshare"] = {"outcome": "ZERO_EXIT", "exit": 0, "termination": "EXITED", "missing": None}
        self.assertFalse(probe.selection_complete(value))
        with self.assertRaises(probe.Refused):
            probe.public_bytes(value)
        value["unshare"] = probe.empty_result()["unshare"]
        value["ldd"]["missing"] = False
        self.assertFalse(probe.selection_complete(value))
        with self.assertRaises(probe.Refused):
            probe.public_bytes(value)

    def test_parser_elf_and_native_guards_are_structurally_unchanged(self):
        base = ast.parse((REFERENCE_DIR / "k6-linux-probes-20261030.py").read_bytes())
        candidate = ast.parse(Path(probe.__file__).read_bytes())
        def function(tree, name):
            return ast.dump(next(n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == name), include_attributes=False)
        for name in ("private_chain", "load_api", "trusted_system", "elf_metadata", "ldd_count", "native_home"):
            self.assertEqual(function(base, name), function(candidate, name))

    def test_parent_file_operations_preserve_the_reviewed_implementations(self):
        canonical = reference.CANON.parents[2] / ".tmp/k6-probe-parent-files-20261030.py"
        base = ast.parse(canonical.read_bytes())
        candidate = ast.parse((HERE / "k6-probe-parent-files-ldd-only-20261030.py").read_bytes())
        functions = lambda tree: {n.name: ast.dump(n, include_attributes=False) for n in tree.body if isinstance(n, ast.FunctionDef)}
        self.assertEqual(functions(base), functions(candidate))
        values = {n.targets[0].id: ast.literal_eval(n.value) for n in candidate.body if isinstance(n, ast.Assign)
                  and isinstance(n.targets[0], ast.Name) and n.targets[0].id in {"PROBE_SHA", "EXPECTED", "DIRS", "RUN"}}
        self.assertEqual(values["EXPECTED"], probe.EXPECTED)
        self.assertEqual(values["PROBE_SHA"], hashlib.sha256(Path(probe.__file__).read_bytes()).hexdigest())
        self.assertEqual(values["RUN"], "ldd-only-20261030")
        self.assertEqual(len(values["EXPECTED"]) + 1, 5)
        parent = (HERE / "k6-probe-parent-ldd-only-20261030.ps1").read_text(encoding="utf8")
        self.assertIn("$ProbeSha='" + values["PROBE_SHA"] + "'", parent)
        bridge_sha = hashlib.sha256((HERE / "k6-probe-parent-files-ldd-only-20261030.py").read_bytes()).hexdigest()
        self.assertIn("$BridgeSha='" + bridge_sha + "'", parent)

    def test_ci_refuses_before_native_guards_or_any_command(self):
        console = io.BytesIO()
        with mock.patch.dict(os.environ, {"CI": ""}, clear=True), mock.patch.object(probe, "native_home") as home, \
                mock.patch.object(probe, "load_api") as api, mock.patch.object(probe, "run_one") as command, \
                mock.patch.object(sys, "stdout", SimpleNamespace(buffer=console)):
            self.assertEqual(probe.main(["--selection", "ldd-only", "--source-dir", "x", "--tools-dir", "x", "--output", "x"]), 2)
        home.assert_not_called(); api.assert_not_called(); command.assert_not_called()
        self.assertEqual(json.loads(console.getvalue())["unshare"], probe.empty_result()["unshare"])


if __name__ == "__main__":
    unittest.main()
